4//! Folder structure:
5//! {base_path}/{network_name}/forked_at_{checkpoint}/
6//!     - objects/
7//!         - {object_id}/
8//!            - latest (contains the latest version number)
9//!            - {version} (contains the object data in BCS format)
10//!      - checkpoints/
11//!          - latest (contains the latest checkpoint sequence number)
12//!          - {checkpoint} (contains the checkpoint data in BCS format)
13//!      - transactions/
14//!          - {tx_digest}/
15//!              - data    (BCS-encoded Transaction envelope)
16//!              - effects (BCS-encoded TransactionEffects)
17//!              - events  (BCS-encoded TransactionEvents)

39/// Directory name appended to the configured filesystem store root.
40const DATA_STORE_DIR: &str = ".forking_data_store";
41/// Per-chain object storage directory.
42const OBJECTS_DIR: &str = "objects";
43/// Per-chain checkpoint storage directory.
44const CHECKPOINTS_DIR: &str = "checkpoints";
45/// Per-chain transaction storage directory.
46const TRANSACTIONS_DIR: &str = "transactions";
47/// Filename for the BCS-encoded transaction data within a transaction directory.
48const TX_DATA_FILE: &str = "data";
49/// Filename for the BCS-encoded transaction effects within a transaction directory.
50const TX_EFFECTS_FILE: &str = "effects";
51/// Filename for the BCS-encoded transaction events within a transaction directory.
52const TX_EVENTS_FILE: &str = "events";
53/// Marker file for the latest checkpoint sequence known to the store.
54const LATEST_FILE: &str = "latest";
55
56/// Local filesystem-backed store for Sui data.
57pub(crate) struct FilesystemStore {
58    root: PathBuf,
59}
60
61impl FilesystemStore {
62    /// Create a new filesystem store rooted at
63    /// `{base_path}/{network_name}/forked_at_{checkpoint}`.
64    pub(crate) fn new(
65        node: &Node,
66        forked_at_checkpoint: CheckpointSequenceNumber,
67    ) -> Result<Self, Error> {
68        let root = Self::base_path()?
69            .join(node.network_name())
70            .join(format!("forked_at_{}", forked_at_checkpoint));
71        Ok(Self { root })
72    }
73
74    /// Create a filesystem store with an explicit root directory.
75    #[cfg(test)]
76    pub(crate) fn new_with_root(root: PathBuf) -> Self {
77        Self { root }
78    }

80    /// Resolve the default base path for on-disk storage.
81    pub(crate) fn base_path() -> Result<PathBuf, Error> {
82        let home_dir = std::env::var("FORKING_DATA_STORE")
83            .or_else(|_| std::env::var("SUI_CONFIG_DIR"))
84            .or_else(|_| std::env::var("HOME"))
85            .or_else(|_| std::env::var("USERPROFILE"))
86            .map_err(|_| {
87                anyhow!(
88                    "Cannot determine home directory. Define a FORKING_DATA_STORE environment variable"
89                )
90            })?;
91        Ok(PathBuf::from(home_dir).join(DATA_STORE_DIR))
92    }
93
94    /// Return the directory path for storing objects data.
95    fn objects_dir(&self) -> PathBuf {
96        self.root.join(OBJECTS_DIR)
97    }
98
99    /// Return the directory path for storing checkpoint data.
100    fn checkpoints_dir(&self) -> PathBuf {
101        self.root.join(CHECKPOINTS_DIR)
102    }
103
104    /// Return the directory path for storing transaction data.
105    fn transactions_dir(&self) -> PathBuf {
106        self.root.join(TRANSACTIONS_DIR)
107    }
108
109    /// Return the directory for a specific transaction.
110    fn transaction_dir(&self, digest: &TransactionDigest) -> PathBuf {
111        self.transactions_dir().join(digest.to_string())
112    }

114    /// Persist a verified transaction to disk under `transactions/{digest}/data`. The underlying
115    /// `Transaction` envelope is what gets serialized; the verified marker is reapplied on read.
116    pub(crate) fn write_transaction(
117        &self,
118        digest: &TransactionDigest,
119        transaction: &VerifiedTransaction,
120    ) -> Result<(), Error> {
121        let path = self.transaction_dir(digest).join(TX_DATA_FILE);
122        self.write_bcs_file(&path, transaction.inner())
123    }
124
125    /// Persist transaction effects to disk under `transactions/{digest}/effects`.
126    pub(crate) fn write_transaction_effects(
127        &self,
128        digest: &TransactionDigest,
129        effects: &TransactionEffects,
130    ) -> Result<(), Error> {
131        let path = self.transaction_dir(digest).join(TX_EFFECTS_FILE);
132        self.write_bcs_file(&path, effects)
133    }

138        digest: &TransactionDigest,
139        events: &TransactionEvents,

145    /// Read a previously persisted transaction. Returns `None` if the transaction is not on disk.
146    pub(crate) fn get_transaction(
147        &self,
148        digest: &TransactionDigest,
149    ) -> Result<Option<VerifiedTransaction>, Error> {
150        let path = self.transaction_dir(digest).join(TX_DATA_FILE);
151        if !path.exists() {
152            return Ok(None);
153        }
154        let inner: Transaction = self.read_bcs_file(&path)?;
155        Ok(Some(VerifiedTransaction::new_unchecked(inner)))
156    }

158    /// Read previously persisted transaction effects. Returns `None` if not on disk.
159    pub(crate) fn get_transaction_effects(
160        &self,
161        digest: &TransactionDigest,
162    ) -> Result<Option<TransactionEffects>, Error> {
163        let path = self.transaction_dir(digest).join(TX_EFFECTS_FILE);
164        if !path.exists() {
165            return Ok(None);

182    /// Get the latest object version available on disk for the given object ID.
183    pub(crate) fn get_latest_object(&self, object_id: &ObjectID) -> anyhow::Result<Option<Object>> {
184        let object_dir = self.objects_dir().join(object_id.to_string());
185
186        if !object_dir.exists() {
187            return Ok(None);
188        }
189
190        let latest_version = self.read_latest_file(&object_dir)?;
191        let version_file = object_dir.join(latest_version.to_string());
192        self.read_bcs_file(&version_file).map(Some)
193    }
194
195    /// Get the object at the given version for the given object ID. Returns `None` if the version
196    /// file does not exist on disk.
197    pub(crate) fn get_object_at_version(
198        &self,
199        object_id: &ObjectID,
200        version: u64,
201    ) -> anyhow::Result<Option<Object>> {
202        let object_dir = self.objects_dir().join(object_id.to_string());
203        let version_file = object_dir.join(version.to_string());
204
205        if !version_file.exists() {
206            return Ok(None);
207        }
208
209        self.read_bcs_file(&version_file).map(Some)
210    }
211
212    /// Write the given object to disk under the objects directory, using the object ID and version
213    /// as the path. It will also update the latest file to point to this version.
214    pub(crate) fn write_object(&self, object: &Object) -> anyhow::Result<()> {
215        let object_dir = self.objects_dir().join(object.id().to_string());
216        let version = object.version().value();
217        let version_file = object_dir.join(version.to_string());
218        self.write_bcs_file(&version_file, object)?;
219
220        let latest_version = if object_dir.join(LATEST_FILE).exists() {
221            std::cmp::max(self.read_latest_file(&object_dir)?, version)
222        } else {
223            version
224        };
225        let latest_file = object_dir.join(LATEST_FILE);
226        fs::write(latest_file, latest_version.to_string())
227            .with_context(|| format!("Failed to write latest file for object {}", object.id()))
228    }
229
230    /// Get the highest checkpoint sequence number available on disk.
231    pub(crate) fn get_highest_checkpoint_sequence_number(
232        &self,
233    ) -> anyhow::Result<CheckpointSequenceNumber> {
234        let checkpoint_dir = self.checkpoints_dir();
235
236        anyhow::ensure!(
237            checkpoint_dir.exists(),
238            "Checkpoint directory does not exist: {}",
239            checkpoint_dir.display()
240        );
241
242        self.read_latest_file(&checkpoint_dir)
243    }
244
245    fn read_bcs_file<T: serde::de::DeserializeOwned>(&self, path: &Path) -> Result<T, Error> {
246        let bytes =
247            fs::read(path).with_context(|| format!("Failed to read file: {}", path.display()))?;
248        bcs::from_bytes(&bytes)
249            .with_context(|| format!("Failed to deserialize BCS data from: {}", path.display()))
250    }
251
252    fn write_bcs_file<T: serde::Serialize>(&self, path: &Path, data: &T) -> Result<(), Error> {
253        if let Some(parent) = path.parent().filter(|p| !p.as_os_str().is_empty()) {
254            fs::create_dir_all(parent)
255                .with_context(|| format!("Failed to create directory: {}", parent.display()))?;
256        }
257        let bytes = bcs::to_bytes(data)
258            .with_context(|| format!("Failed to serialize BCS data for: {}", path.display()))?;
259        fs::write(path, bytes)
260            .with_context(|| format!("Failed to write file: {}", path.display()))?;
261        Ok(())
262    }
263
264    /// Read the latest file that contains a number representing the latest checkpoint sequence or
265    /// object version.
266    fn read_latest_file(&self, dir: &Path) -> Result<u64, Error> {
267        let latest_path = dir.join(LATEST_FILE);
268        if !latest_path.exists() {
269            bail!("Latest file not found in directory: {}", dir.display());
270        }
271        let content = fs::read_to_string(&latest_path)
272            .with_context(|| format!("Failed to read latest file: {}", latest_path.display()))?;
273        content
274            .trim()
275            .parse::<u64>()
276            .with_context(|| format!("Failed to parse latest file: {}", latest_path.display()))
277    }

294    fn test_store() -> (tempfile::TempDir, FilesystemStore) {

296        let store = FilesystemStore::new_with_root(dir.path().to_path_buf());

300    fn make_object(id: ObjectID, version: u64) -> Object {
301        let move_obj = MoveObject::new_gas_coin(SequenceNumber::from_u64(version), id, 1_000_000);
302        ObjectInner {
303            owner: Owner::Immutable,
304            data: sui_types::object::Data::Move(move_obj),
305            previous_transaction: TransactionDigest::genesis_marker(),
306            storage_rebate: 0,
307        }
308        .into()
309    }

312    fn test_write_and_read_latest_object() {
313        let (_dir, store) = test_store();
314        let id = ObjectID::random();
315        let obj = make_object(id, 5);
316
317        store.write_object(&obj).unwrap();
318        let loaded = store.get_latest_object(&id).unwrap();
319        assert_eq!(loaded.clone().unwrap(), obj);
320        assert_eq!(loaded.unwrap().version(), SequenceNumber::from_u64(5));
321    }
