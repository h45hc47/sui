22macro_rules! block_on {
23    ($expr:expr) => {{
24        #[allow(clippy::disallowed_methods, clippy::result_large_err)]
25        {
26            if tokio::runtime::Handle::try_current().is_ok() {
27                std::thread::scope(|scope| {
28                    scope
29                        .spawn(|| {
30                            let rt = tokio::runtime::Builder::new_current_thread()
31                                .enable_all()
32                                .build()
33                                .expect("failed to build Tokio runtime");
34                            rt.block_on($expr)
35                        })
36                        .join()
37                        .expect("failed to join scoped thread running nested runtime")
38                })
39            } else {
40                let rt = tokio::runtime::Builder::new_current_thread()
41                    .enable_all()
42                    .build()
43                    .expect("failed to build Tokio runtime");
44                rt.block_on($expr)
45            }
46        }
47    }};
48}

50/// GraphQL Client for querying the GraphQL service.
51#[derive(Debug, Clone)]
52pub struct GraphQLClient {
53    client: reqwest::Client,
54    node: Node,
55    rpc: reqwest::Url,
56    version: String,
57}
58
59impl GraphQLClient {
60    /// Create a new GraphQL client
61    pub fn new(node: Node, version: &str) -> Result<Self, Error> {
62        let rpc = reqwest::Url::parse(node.gql_url())
63            .with_context(|| format!("invalid GraphQL URL '{}'", node.gql_url()))?;
64        Ok(Self {
65            client: reqwest::Client::new(),
66            node,
67            rpc,
68            version: version.to_string(),
69        })
70    }
71
72    pub(crate) async fn run_query<T, V>(
73        &self,
74        operation: &Operation<T, V>,
75    ) -> Result<GraphQlResponse<T>, Error>
76    where
77        T: serde::de::DeserializeOwned,
78        V: serde::Serialize,
79    {
80        Self::run_query_internal(&self.client, &self.rpc, &self.version, operation).await
81    }
82
83    async fn run_query_internal<T, V>(
84        client: &reqwest::Client,
85        rpc: &reqwest::Url,
86        version: &str,
87        operation: &Operation<T, V>,
88    ) -> Result<GraphQlResponse<T>, Error>
89    where
90        T: serde::de::DeserializeOwned,
91        V: serde::Serialize,
92    {
93        client
94            .post(rpc.clone())
95            .header(USER_AGENT, format!("sui-forking-v{}", version))
96            .json(operation)
97            .send()
98            .await
99            .context("Failed to send GQL query")?
100            .json::<GraphQlResponse<T>>()
101            .await
102            .context("Failed to read response in GQL query")
103    }
104
105    async fn get_verified_checkpoint_impl(
106        &self,
107        sequence_number: Option<CheckpointSequenceNumber>,
108    ) -> Result<Option<VerifiedCheckpoint>, Error> {
109        queries::checkpoint_query::query(sequence_number, self).await
110    }
111
112    /// Get the latest checkpoint sequence number from GraphQL RPC.
113    pub async fn get_latest_checkpoint_sequence_number(
114        &self,
115    ) -> Result<Option<CheckpointSequenceNumber>, Error> {
116        queries::latest_checkpoint_query::query(self).await
117    }
118
119    pub(crate) fn chain(&self) -> Chain {
120        match self.node {
121            Node::Mainnet => Chain::Mainnet,
122            Node::Testnet => Chain::Testnet,
123            Node::Devnet => Chain::Unknown,
124            Node::Custom(_) => Chain::Unknown,
125        }
126    }
127}

138impl ObjectRead for GraphQLClient {
139    fn get_objects(&self, keys: &[ObjectKey]) -> Result<Vec<Option<(Object, u64)>>, Error> {
140        block_on!(crate::gql::queries::object_query::query(keys, self))
141    }
142}
143
144impl CheckpointRead for GraphQLClient {
145    fn get_verified_checkpoint(
146        &self,
147        sequence: Option<CheckpointSequenceNumber>,
148    ) -> Result<Option<VerifiedCheckpoint>, Error> {
149        Ok(block_on!(self.get_verified_checkpoint_impl(sequence))?)
150    }

170    fn checkpoint_response_body(
171        certified: &sui_types::messages_checkpoint::CertifiedCheckpointSummary,
172    ) -> serde_json::Value {
173        json!({
174            "data": {
175                "checkpoint": {
176                    "summaryBcs": FastCryptoBase64::from_bytes(
177                        &bcs::to_bytes(certified.data()).expect("summary should serialize"),
178                    )
179                    .encoded(),
180                    "validatorSignatures": {
181                        "signature": FastCryptoBase64::from_bytes(
182                            certified.auth_sig().signature.as_ref(),
183                        )
184                        .encoded(),
185                        "signersMap": certified
186                            .auth_sig()
187                            .signers_map
188                            .iter()
189                            .map(|index| i32::try_from(index).expect("signer index fits in i32"))
190                            .collect::<Vec<_>>(),
191                    },
192                }
193            }
194        })
195    }
196
197    fn object_response_body(objects: &[Option<&Object>]) -> serde_json::Value {
198        json!({
199            "data": {
200                "multiGetObjects": objects
201                    .iter()
202                    .map(|object| {
203                        object.as_ref().map(|object| {
204                            json!({
205                                "address": object.id().to_string(),
206                                "version": object.version().value(),
207                                "objectBcs": FastCryptoBase64::from_bytes(
208                                    &bcs::to_bytes(*object).expect("object should serialize"),
209                                )
210                                .encoded(),
211                            })
212                        })
213                    })
214                    .collect::<Vec<_>>(),
215            }
216        })
217    }
218
219    fn versioned_object_at_checkpoint_response_body(object: Option<&Object>) -> serde_json::Value {
220        json!({
221            "data": {
222                "checkpoint": {
223                    "query": {
224                        "object": object.map(|object| {
225                            json!({
226                                "address": object.id().to_string(),
227                                "version": object.version().value(),
228                                "objectBcs": FastCryptoBase64::from_bytes(
229                                    &bcs::to_bytes(object).expect("object should serialize"),
230                                )
231                                .encoded(),
232                            })
233                        })
234                    }
235                }
236            }
237        })
238    }

329        let server = MockServer::start().await;
330        let versioned_object = Object::immutable_with_id_for_testing(ObjectID::random());

339                    "keys": [
340                        {
341                            "address": versioned_object.id().to_string(),
342                            "version": versioned_object.version().value(),
343                        },
344                        {
345                            "address": root_version_object.id().to_string(),
346                            "rootVersion": 17,
347                        },
348                        {
349                            "address": missing_object_id.to_string(),
350                            "atCheckpoint": 29,
351                        },
352                    ],
353                }
354            })))
355            .respond_with(
356                ResponseTemplate::new(200).set_body_json(object_response_body(&[
357                    Some(&versioned_object),
358                    Some(&root_version_object),
359                    None,
360                ])),
361            )
362            .mount(&server)
363            .await;
364
365        let store = mock_store(&server);
366        let objects = store
367            .get_objects(&[

370                    version_query: VersionQuery::Version(versioned_object.version().value()),

372                ObjectKey {
373                    object_id: root_version_object.id(),
374                    version_query: VersionQuery::RootVersion(17),
375                },

381            .expect("object query should succeed");
382
383        assert_eq!(

387                Some((
388                    root_version_object.clone(),
389                    root_version_object.version().value(),
390                )),
391                None,
392            ]
393        );
394
395        let requests = server
396            .received_requests()
397            .await
398            .expect("wiremock should record requests");
399        let request_body: serde_json::Value = requests[0]
400            .body_json()
401            .expect("request body should be json");
402        let query = request_body
403            .get("query")
404            .and_then(serde_json::Value::as_str)
405            .expect("query string should be present");
